// Audit only: each probe uses a disposable test database.
func TestAuditCommitFailureConnectionState(t *testing.T) {
 db,err:=Open(filepath.Join(t.TempDir(),"audit.db"));if err!=nil {t.Fatal(err)};defer db.Close()
 tx,err:=db.Begin();if err!=nil {t.Fatal(err)}
 if _,err=tx.Exec(`PRAGMA defer_foreign_keys=ON; INSERT INTO plans(subject_id,start_time,duration_minutes) VALUES(999,'09:00',30)`);err!=nil {t.Fatal(err)}
 commitErr:=tx.Commit();if commitErr==nil {t.Fatal("expected deferred FK commit error")}
 rollbackErr:=tx.Rollback();if !errors.Is(rollbackErr,sql.ErrTxDone) {t.Fatalf("rollback error = %v",rollbackErr)}
 next,beginErr:=db.Begin();if beginErr==nil {next.Rollback();t.Fatal("expected leaked SQLite transaction")}
 t.Logf("AUDIT: Commit=%v; deferred Rollback=%v; next Begin=%v",commitErr,rollbackErr,beginErr)
 // Explicit cleanup of this disposable database only.
 if _,err=db.Exec(`ROLLBACK`);err!=nil {t.Fatal(err)}
}
func TestAuditSubjectArchivePartialSuccess(t *testing.T) {
 db,err:=Open(filepath.Join(t.TempDir(),"audit.db"));if err!=nil {t.Fatal(err)};defer db.Close()
 _,err=db.Exec(`INSERT INTO subjects(id,name,created_at) VALUES(1,'audit','2026-11-18');
 INSERT INTO plans(id,subject_id,start_time,duration_minutes) VALUES(1,1,'09:00',30);
 CREATE TRIGGER reject_plan_state BEFORE UPDATE OF active ON plans BEGIN SELECT RAISE(ABORT,'injected plan update failure'); END;`)
 if err!=nil {t.Fatal(err)}
 _,err=subjects.Update(db,subjects.Subject{ID:1,Name:"audit",Archived:true})
 if err!=nil {t.Fatalf("expected swallowed error, got %v",err)}
 var archived,active int
 if err=db.QueryRow(`SELECT archived,(SELECT active FROM plans WHERE id=1) FROM subjects WHERE id=1`).Scan(&archived,&active);err!=nil {t.Fatal(err)}
 if archived!=1||active!=1 {t.Fatalf("unexpected states %d %d",archived,active)}
 t.Log("AUDIT: archive reported success; subject archived=1, plan active=1 after injected SQL failure")
}

func TestAuditSingleConnectionDoesNotSerializeBusiness(t *testing.T) {
 db,err:=Open(filepath.Join(t.TempDir(),"audit.db"));if err!=nil {t.Fatal(err)};defer db.Close()
 // Show that holding an open Tx prevents db-level calls; prevents accidental fix using db inside tx.
 tx,err:=db.Begin();if err!=nil {t.Fatal(err)};defer tx.Rollback()
 ctx,cancel:=context.WithTimeout(context.Background(),20*time.Millisecond);defer cancel()
 _,err=db.ExecContext(ctx,`SELECT 1`)
 if !errors.Is(err,context.DeadlineExceeded) {t.Fatalf("unexpected wait result=%v",err)}
 t.Log("AUDIT: db.Exec waits for the only connection held by tx; all in-transaction helpers must accept tx")
}

func TestAuditActualStartCommitBusyLeavesTransaction(t *testing.T) {
 path:=filepath.Join(t.TempDir(),"audit.db")
 db,err:=Open(path);if err!=nil {t.Fatal(err)};defer db.Close()
 if _,err=db.Exec(`INSERT INTO subjects(id,name,created_at) VALUES(1,'audit','2026-11-18')`);err!=nil {t.Fatal(err)}
 reader,err:=OpenExisting(path);if err!=nil {t.Fatal(err)};defer reader.Close()
 readTx,err:=reader.Begin();if err!=nil {t.Fatal(err)};defer readTx.Rollback()
 var n int
 if err=readTx.QueryRow(`SELECT count(*) FROM subjects`).Scan(&n);err!=nil {t.Fatal(err)}
 _,commitErr:=sessions.Start(db,sessions.TimerInput{SubjectID:1},time.Date(2026,11,18,9,0,0,0,time.Local))
 if commitErr==nil {t.Fatal("expected commit lock error")}
 next,beginErr:=db.Begin();if beginErr==nil {next.Rollback();t.Fatal("expected residual SQLite transaction")}
 t.Logf("AUDIT actual Start: external reader -> Commit=%v; next Begin=%v",commitErr,beginErr)
 if err=readTx.Rollback();err!=nil {t.Fatal(err)}
 if _,err=db.Exec(`ROLLBACK`);err!=nil {t.Fatal(err)}
}
